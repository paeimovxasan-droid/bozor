# ── 1-USUL: CLI cron (eng ishonchli, ISPmanager "Cron jobs") ──
# Yo'l: ISPmanager → Vazifalar → Cron jobs → Yaratish
* * * * * /usr/bin/php /home/USER/www/domen.uz/tortinmang/cycle.php >> /dev/null 2>&1
#
# php yo'li hostingda farq qilishi mumkin: /usr/local/bin/php,
# /opt/php84/bin/php ... ISPmanager cron muharririda ko'pincha
# faqat: php /home/USER/... deb yozish ham kifoya.
#
# ── 2-USUL: HTTP cron (wget/curl) — CLI yo'q bo'lsa ────────────
# CRON_KEY ni install.php saqlanganda ko'rsatilgan URL dan oling:
* * * * * wget -q -O /dev/null "https://domen.uz/tortinmang/cycle.php?key=CRON_KEY"
# yoki:
* * * * * curl -s -o /dev/null "https://domen.uz/tortinmang/cycle.php?key=CRON_KEY"
